3 Three Dimensional Scatterplot / Spaghetti Plot

Code 
clear3d()

scatter3d(simulated_multilevel_longitudinal_data$outcome ~ 
            simulated_multilevel_longitudinal_data$warmth +
            simulated_multilevel_longitudinal_data$t,
          groups = simulated_multilevel_longitudinal_data$id,
          surface.col = 1:30,
          surface = TRUE,
          xlab="x",
          zlab = "t",
          ylab = "Outcome")

text3d(x=0, y=1, z=0, 
       "Three Dimensional Scatterplot & Spaghetti Plot",
       cex = 2.0)

rglwidget()
